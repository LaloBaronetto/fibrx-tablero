def fetch_market_rate(vehiculo: dict) -> float:
    """
    Obtiene la TNA actual según el tipo de fuente configurada.
    """
    stype = vehiculo["source_type"]
    try:
        if stype == "IOL_GATEWAY":
            return fetch_iol_rates()
        # Aquí se pueden agregar conectores específicos para BYMA, MAE o APIs de AlyCs.
        # Si falla la conexión externa, el sistema no se rompe: utiliza el fallback de seguridad.
        return vehiculo["tna_fallback"]
    except Exception as e:
        print(f"[WARN] Error al actualizar {vehiculo['id']}: {e}. Usando fallback.")
        return vehiculo["tna_fallback"]

def main():
    print(f"[*] Iniciando ingesta de tasas Fibrx — {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}")
    
    output_data = {
        "actualizado_el": datetime.now().isoformat(),
        "vehiculos": []
    }

    for v in VEHICULOS_CONFIG:
        tna_actual = fetch_market_rate(v)
        registro = {
            "id": v["id"],
            "nombre": v["nombre"],
            "tna": tna_actual,
            "fees": v["fees"],
            "min": v["min"],
            "liq": v["liq"],
            "tax": v["tax"],
            "risky": v["risky"],
            "cur": v["cur"]
        }
        output_data["vehiculos"].append(registro)
        print(f"  -> {v['id']}: TNA {tna_actual*100:.2f}%")

    # Guardar JSON consolidado para consumo del frontend o de la v2
    nombre_archivo = "tasas_mercado.json"
    with open(nombre_archivo, "w", encoding="utf-8") as f:
        json.dump(output_data, f, indent=2, ensure_ascii=False)
    
    print(f"[✔] Ingesta completada. Archivo generado: {nombre_archivo}")

if __name__ == "__main__":
    main()
